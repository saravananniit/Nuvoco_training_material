import hashlib

# Global variable
stored_name = None


def generate_encrypted_text(name: str) -> str:
    global stored_name
    stored_name = name

    return hashlib.sha256(name.encode("utf-8")).hexdigest()


def validate_name(encrypted_text: str) -> str:
    global stored_name

    if stored_name is None:
        return "Error: No name stored."

    regenerated_hash = hashlib.sha256(
        stored_name.encode("utf-8")
    ).hexdigest()

    if regenerated_hash == encrypted_text:
        return stored_name

    return "Error: Invalid encrypted text."


# Generate
name = input("Enter Name: ")
encrypted = generate_encrypted_text(name)

print("\nEncrypted Text:")
print(encrypted)

# Validate
user_encrypted = input("\nPaste Encrypted Text: ")

result = validate_name(user_encrypted)
print("\nResult:", result)
